    # ======================================================
    # INITIAL CLIENT INFORMATION
    # ======================================================

    def handle_client_info(
        self,
        client_id,
        client_list
    ):

        self.my_client_id = client_id


        self.my_id_label.setText(
            f"Client {client_id}"
        )


        self.online_clients = list(
            client_list
        )


        # --------------------------------------------------
        # Create conversation containers
        # --------------------------------------------------

        self.ensure_conversation(
            0
        )


        for client in client_list:

            self.ensure_conversation(
                client
            )


        # --------------------------------------------------
        # Update sidebar
        # --------------------------------------------------

        self.update_client_list(
            client_list
        )


    # ======================================================
    # UPDATE CLIENT LIST
    # ======================================================

    def update_client_list(
        self,
        client_ids
    ):

        # --------------------------------------------------
        # Remove duplicates
        # --------------------------------------------------

        clean_ids = []

        for client_id in client_ids:

            if client_id not in clean_ids:

                clean_ids.append(
                    client_id
                )


        self.online_clients = clean_ids


        # --------------------------------------------------
        # Save current selection
        # --------------------------------------------------

        current_selection = (
            self.selected_client_id
        )


        # --------------------------------------------------
        # Make sure conversations exist
        # --------------------------------------------------

        self.ensure_conversation(
            0
        )


        for client_id in clean_ids:

            self.ensure_conversation(
                client_id
            )


        # --------------------------------------------------
        # Rebuild list
        # --------------------------------------------------

        self.client_list.clear()


        # ==================================================
        # EVERYONE
        # ==================================================

        everyone_item = QListWidgetItem(
            self.build_everyone_text()
        )


        everyone_item.setData(
            Qt.UserRole,
            0
        )


        self.client_list.addItem(
            everyone_item
        )


        # ==================================================
        # CLIENTS
        # ==================================================

        # Show only OTHER clients here.
        # Our own client identity is already shown in the user card.

        for client_id in clean_ids:

            if client_id == self.my_client_id:
                continue

            text = self.build_client_text(
                client_id,
                False
            )

            item = QListWidgetItem(
                text
            )

            item.setData(
                Qt.UserRole,
                client_id
            )

            self.client_list.addItem(
                item
            )

        # --------------------------------------------------
        # Restore previous conversation
        # --------------------------------------------------

        if current_selection == 0:

            self.selected_client_id = 0

            self.select_list_item(
                0
            )

            self.set_broadcast_header()


        elif (
            current_selection in clean_ids
            and current_selection != self.my_client_id
        ):

            self.selected_client_id = (
                current_selection
            )


            self.select_list_item(
                current_selection
            )


            self.set_private_header(
                current_selection
            )


        else:

            # Selected client disappeared

            self.selected_client_id = 0


            self.select_list_item(
                0
            )


            self.set_broadcast_header()


        # --------------------------------------------------
        # Redraw current conversation
        # --------------------------------------------------

        self.show_conversation(
            self.selected_client_id
        )


    # ======================================================
    # BUILD EVERYONE SIDEBAR TEXT
    # ======================================================

    def build_everyone_text(
        self
    ):

        unread = self.unread_counts.get(
            0,
            0
        )


        last = self.last_messages.get(
            0,
            ""
        )


        if last:

            preview = self.truncate(
                last,
                25
            )

            text = (
                f"🌐   Everyone\n"
                f"     {preview}"
            )

        else:

            text = (
                "🌐   Everyone\n"
                "     Broadcast chat"
            )


        if unread > 0:

            text += (
                f"   • {unread}"
            )


        return text


    # ======================================================
    # BUILD CLIENT SIDEBAR TEXT
    # ======================================================

    def build_client_text(
        self,
        client_id,
        is_me
    ):

        if is_me:

            icon = "🟢"

            name = (
                f"Client {client_id} (You)"
            )

        else:

            icon = "🟢"

            name = (
                f"Client {client_id}"
            )


        last = self.last_messages.get(
            client_id,
            ""
        )


        unread = self.unread_counts.get(
            client_id,
            0
        )


        if last:

            preview = self.truncate(
                last,
                25
            )

            text = (
                f"{icon}   {name}\n"
                f"     {preview}"
            )

        else:

            text = (
                f"{icon}   {name}"
            )


        if unread > 0:

            text += (
                f"   • {unread}"
            )


        return text


    # ======================================================
    # TRUNCATE TEXT
    # ======================================================

    def truncate(
        self,
        text,
        length
    ):

        text = text.replace(
            "\n",
            " "
        )


        if len(text) <= length:

            return text


        return (
            text[:length - 3]
            + "..."
        )


    # ======================================================
    # SELECT CLIENT
    # ======================================================

    def select_client(
        self,
        item
    ):

        client_id = item.data(
            Qt.UserRole
        )


        # ==================================================
        # EVERYONE
        # ==================================================

        if client_id == 0:

            self.selected_client_id = 0


            self.unread_counts[0] = 0


            self.set_broadcast_header()


            self.show_conversation(
                0
            )


            self.refresh_sidebar(
                preserve_selection=True
            )


            self.message_input.setFocus()

            return


        # ==================================================
        # OWN CLIENT
        # ==================================================

        if client_id == self.my_client_id:

            # Don't open private chat with yourself

            self.selected_client_id = 0


            self.select_list_item(
                0
            )


            self.set_broadcast_header()


            self.show_conversation(
                0
            )


            self.message_input.setFocus()

            return


        # ==================================================
        # PRIVATE CLIENT
        # ==================================================

        self.selected_client_id = client_id


        self.unread_counts[client_id] = 0


        self.set_private_header(
            client_id
        )


        self.show_conversation(
            client_id
        )


        self.refresh_sidebar(
            preserve_selection=True
        )


        self.message_input.setFocus()

    # ======================================================
    # SEND MESSAGE
    # ======================================================

    def send_message(
        self
    ):

        message = (
            self.message_input.text().strip()
        )


        if not message:

            return


        if not self.network.connected:

            self.add_error_message(
                "Not connected to the server."
            )

            return


        success = False


        # ==================================================
        # PRIVATE
        # ==================================================

        if (
            self.selected_client_id != 0
            and self.selected_client_id != self.my_client_id
        ):

            target_id = (
                self.selected_client_id
            )


            success = (
                self.network.send_private_message(
                    target_id,
                    message
                )
            )


            if success:

                self.add_message_to_conversation(
                    target_id,
                    "sent",
                    message,
                    f"You → Client {target_id}"
                )


                self.last_messages[target_id] = (
                    message
                )


        # ==================================================
        # BROADCAST
        # ==================================================

        else:

            success = (
                self.network.send_broadcast_message(
                    message
                )
            )


            if success:

                self.add_message_to_conversation(
                    0,
                    "sent",
                    message,
                    "You → Everyone"
                )


                self.last_messages[0] = (
                    message
                )


        # ==================================================
        # SUCCESS
        # ==================================================

        if success:

            self.message_input.clear()


            self.show_conversation(
                self.selected_client_id
            )


            self.refresh_sidebar(
                preserve_selection=True
            )


        else:

            self.add_error_message(
                "Message could not be sent."
            )


    # ======================================================
    # RECEIVE MESSAGE
    # ======================================================

    def display_received_message(
        self,
        message
    ):

        # ==================================================
        # PRIVATE MESSAGE
        # ==================================================

        if message.startswith(
            "[Private from Client "
        ):

            sender_id = (
                self.extract_private_sender(
                    message
                )
            )


            if sender_id is None:

                self.add_error_message(
                    "Could not identify private-message sender."
                )

                return


            clean_message = (
                self.extract_private_text(
                    message
                )
            )


            # --------------------------------------------------
            # Store under sender's conversation
            # --------------------------------------------------

            self.add_message_to_conversation(
                sender_id,
                "received",
                clean_message,
                f"Client {sender_id}"
            )


            self.last_messages[sender_id] = (
                clean_message
            )


            # --------------------------------------------------
            # If this conversation isn't open,
            # increase unread count.
            # --------------------------------------------------

            if self.selected_client_id != sender_id:

                self.unread_counts[sender_id] = (
                    self.unread_counts.get(
                        sender_id,
                        0
                    ) + 1
                )


            # --------------------------------------------------
            # Update current view
            # --------------------------------------------------

            if self.selected_client_id == sender_id:

                self.show_conversation(
                    sender_id
                )


            self.refresh_sidebar(
                preserve_selection=True
            )


            return


        # ==================================================
        # BROADCAST MESSAGE
        # ==================================================

        if message.startswith(
            "[Broadcast from Client "
        ):

            sender_id = (
                self.extract_broadcast_sender(
                    message
                )
            )


            clean_message = (
                self.extract_broadcast_text(
                    message
                )
            )


            self.add_message_to_conversation(
                0,
                "received",
                clean_message,
                (
                    f"Client {sender_id}"
                    if sender_id is not None
                    else "Client"
                )
            )


            self.last_messages[0] = (
                clean_message
            )


            if self.selected_client_id != 0:

                self.unread_counts[0] = (
                    self.unread_counts.get(
                        0,
                        0
                    ) + 1
                )


            else:

                self.unread_counts[0] = 0


            if self.selected_client_id == 0:

                self.show_conversation(
                    0
                )


            self.refresh_sidebar(
                preserve_selection=True
            )


            return


        # ==================================================
        # SERVER MESSAGE
        # ==================================================

        # Don't clutter conversation with protocol/system
        # messages such as CLIENT_LIST or CLIENT_INFO.

        if (
            message.startswith("CLIENT_LIST|")
            or message.startswith("CLIENT_INFO|")
        ):

            return


        # Server messages are shown as a small system notice.

        self.add_system_message(
            message
        )


    # ======================================================
    # EXTRACT PRIVATE SENDER
    # ======================================================

    def extract_private_sender(
        self,
        message
    ):

        # Expected:
        #
        # [Private from Client 1] Hello

        prefix = (
            "[Private from Client "
        )


        try:

            start = len(prefix)

            end = message.index(
                "]",
                start
            )


            return int(
                message[start:end]
            )


        except Exception:

            return None


    # ======================================================
    # EXTRACT PRIVATE TEXT
    # ======================================================

    def extract_private_text(
        self,
        message
    ):

        try:

            return message.split(
                "] ",
                1
            )[1]

        except:

            return message


    # ======================================================
    # EXTRACT BROADCAST SENDER
    # ======================================================

    def extract_broadcast_sender(
        self,
        message
    ):

        # Expected:
        #
        # [Broadcast from Client 2] Hello

        prefix = (
            "[Broadcast from Client "
        )


        try:

            start = len(prefix)

            end = message.index(
                "]",
                start
            )


            return int(
                message[start:end]
            )


        except Exception:

            return None


    # ======================================================
    # EXTRACT BROADCAST TEXT
    # ======================================================

    def extract_broadcast_text(
        self,
        message
    ):

        try:

            return message.split(
                "] ",
                1
            )[1]

        except:

            return message


    # ======================================================
    # ENSURE CONVERSATION
    # ======================================================

    def ensure_conversation(
        self,
        conversation_id
    ):

        if conversation_id not in self.conversations:

            self.conversations[
                conversation_id
            ] = []


        if conversation_id not in self.unread_counts:

            self.unread_counts[
                conversation_id
            ] = 0


    # ======================================================
    # ADD MESSAGE
    # ======================================================

    def add_message_to_conversation(
        self,
        conversation_id,
        message_type,
        message,
        sender
    ):

        self.ensure_conversation(
            conversation_id
        )


        self.conversations[
            conversation_id
        ].append(
            {
                "type": message_type,
                "message": message,
                "sender": sender
            }
        )


    # ======================================================
    # SHOW CONVERSATION
    # ======================================================

    def show_conversation(
        self,
        conversation_id
    ):

        self.ensure_conversation(
            conversation_id
        )


        self.chat_display.clear()


        messages = self.conversations[
            conversation_id
        ]


        for item in messages:

            if item["type"] == "sent":

                self.chat_display.append(
                    self.format_sent_message(
                        item["sender"],
                        item["message"]
                    )
                )

            else:

                self.chat_display.append(
                    self.format_received_message(
                        item["sender"],
                        item["message"]
                    )
                )


        self.scroll_to_bottom()


    # ======================================================
    # SENT MESSAGE
    # ======================================================

    def format_sent_message(
        self,
        sender,
        message
    ):

        return f'''
        <table width="100%" cellspacing="0" cellpadding="0">

    # ======================================================
    # REFRESH SIDEBAR
    # ======================================================

    def refresh_sidebar(
        self,
        preserve_selection=True
    ):

        current_selection = (
            self.selected_client_id
        )


        self.client_list.clear()


        # ==================================================
        # EVERYONE
        # ==================================================

        everyone_item = QListWidgetItem(
            self.build_everyone_text()
        )


        everyone_item.setData(
            Qt.UserRole,
            0
        )


        self.client_list.addItem(
            everyone_item
        )


        # ==================================================
        # CLIENTS
        # ==================================================

        # Show only OTHER clients here.
        # Our own client identity is already shown in the user card.

        for client_id in self.online_clients:

            if client_id == self.my_client_id:
                continue

            text = self.build_client_text(
                client_id,
                False
            )

            item = QListWidgetItem(
                text
            )

            item.setData(
                Qt.UserRole,
                client_id
            )

            self.client_list.addItem(
                item
            )

        # ==================================================
        # RESTORE SELECTION
        # ==================================================

        if preserve_selection:

            self.select_list_item(
                current_selection
            )


    # ======================================================
    # SELECT LIST ITEM
    # ======================================================

    def select_list_item(
        self,
        client_id
    ):

        for index in range(
            self.client_list.count()
        ):

            item = self.client_list.item(
                index
            )


            if item.data(
                Qt.UserRole
            ) == client_id:

                self.client_list.setCurrentItem(
                    item
                )

                return


    # ======================================================
    # CONNECTION STATUS
    # ======================================================

    def update_connection_status(
        self,
        connected
    ):

        self.connected = connected


        if connected:

            self.status_label.setText(
                "●  Connected"
            )


            self.status_label.setStyleSheet(
                f"color: {SUCCESS};"
            )


        else:

            self.status_label.setText(
                "●  Disconnected"
            )


            self.status_label.setStyleSheet(
                f"color: {DANGER};"
            )


    # ======================================================
    # SCROLL TO BOTTOM
    # ======================================================

    def scroll_to_bottom(
        self
    ):

        scrollbar = (
            self.chat_display.verticalScrollBar()
        )


        scrollbar.setValue(
            scrollbar.maximum()
        )


    # ======================================================
    # HTML ESCAPE
    # ======================================================

    def escape_html(
        self,
        text
    ):

        return (
            text
            .replace("&", "&amp;")
            .replace("<", "&lt;")
            .replace(">", "&gt;")
            .replace('"', "&quot;")
        )

    # CLOSE WINDOW
    # ======================================================

    def closeEvent(
        self,
        event
    ):

        self.network.disconnect()

        event.accept()
